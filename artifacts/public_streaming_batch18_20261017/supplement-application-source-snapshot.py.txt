"""Apply the previously frozen descriptive method to both immutable outcomes."""
from pathlib import Path
import hashlib
import json
import platform
import subprocess
import sys
import numpy as np
import pandas as pd

base = Path(__file__).resolve().parent.parent
root = base / 'work/orderflow_edge_lab'
sys.path.insert(0, str(root / 'src'))
from orderflow_edge_lab import public_streaming_batch18 as frozen
from orderflow_edge_lab import public_screen_supplement as method

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

target = base / 'outputs/Streaming-batch18-descriptive-supplement.json'
assert not target.exists(), 'retain previous application'
assert len(sys.argv) == 2 and len(sys.argv[1]) == 64, 'supply completed immutable result hash'
result_path = base / 'outputs/Streaming-batch18-results.json'
assert sha(result_path) == sys.argv[1]
original = json.loads(result_path.read_text())
source = root / 'src/orderflow_edge_lab/public_streaming_batch18.py'
assert sha(source) == original['evaluator_source_sha256'] == 'c88ca0b8ac93c73cec3adea572378e8d999ad9fac974f8b3108eff56219066db'
config_path = root / 'config/public_streaming_batch18_v1.json'
config = frozen.checked_spec(root, config_path)
review_path = base / 'outputs/Public-screen-supplement-draft-independent-review.json'
review = json.loads(review_path.read_text())
assert review['status'] == 'PASS_DESCRIPTIVE_DRAFT_METHOD_BEFORE_RAW_APPLICATION' and not review['blocking_findings']
bindings = dict(draft_evaluator_exact_sha256=root / 'src/orderflow_edge_lab/public_screen_supplement.py',
                draft_test_exact_sha256=root / 'tests/test_public_screen_supplement.py',
                method_exact_sha256=base / 'outputs/Public-screen-descriptive-supplement-method-draft.md')
for key, path in bindings.items():
    assert sha(path) == review[key], key
runtime = dict(python=platform.python_version(), numpy=np.__version__, pandas=pd.__version__, platform=sys.platform)
assert all(runtime[key] == config['runtime'][key] for key in runtime)
panel, evidence = frozen.shared.retained_panel(root, base / 'outputs/Minute-continuous-source-data',
                                              base / 'work/protective_batch5_cache', config)
assert evidence == original['source']

def population(events, paths, calendar):
    result = method.event_summary(events, calendar)
    fields = ('pre_first_mfe_mae_bps_bounds', 'mae_through_endpoint_bps_bounds',
              'hypothetical_peak_net_after_cost_bps_bounds', 'mfe_giveback_to_endpoint_bps_bounds')
    result['path_metrics'] = {
        key: {stat:[function([p[key][i] for p in paths]) for i in (0,1)]
              for stat,function in [('mean',method.average),('median',method.middle)]}
        for key in fields}
    result['path_fractions'] = {key:sum(p[key] for p in paths)/len(paths) if paths else None
                               for key in ('endpoint_correct_direction','definite_positive_excursion','possible_positive_excursion')}
    result['hypothetical_peak_clears_baseline_cost_fraction_bounds'] = [
        sum(p['hypothetical_peak_net_after_cost_bps_bounds'][i]>0 for p in paths)/len(paths) if paths else None
        for i in (0,1)]
    return result

reports = []
for candidate in original['results']:
    assert candidate['decision'] in ('REJECTED_AS_ADAPTED','SCREEN_SURVIVOR_FOR_FUTURE_REGISTRATION')
    case = candidate['cost_and_gap_cases']['40:conservative_limit']
    assert len(case['joint_histories']) == 1 and all(len(states)==1 for states in case['symbols'].values())
    events = [e for states in case['symbols'].values() for e in states[0]['events']]
    assert len(events) == case['joint_histories'][0]['summary']['admitted']
    completed = [i for i,e in enumerate(events) if e['status']=='COMPLETED']
    assert len(completed) == case['joint_histories'][0]['summary']['completed']
    detail = [method.excursion_details(e,panel[e['symbol']]) for e in events]
    calendar = case['joint_histories'][0]['diagnostics']['calendar_weeks_with_zeros']
    assert calendar == 172
    clocks = {}
    for symbol in config['symbols']:
        frame = frozen.shared.bars8h(panel[symbol])
        feature = frozen.source_features(frame,candidate['candidate'],config)
        clock = method.source_signal_clock(frame,feature['long'],config['warmup_bars'])
        expected = case['symbols'][symbol][0]['counts'].get('raw_entry_flags',0)
        assert clock['completion_clock']['n'] == expected,symbol
        clocks[symbol] = dict(raw_entry_flags=expected,**clock)
    reports.append(dict(candidate=candidate['candidate'],original_decision=candidate['decision'],
                        original_failed_gates=candidate['failed_gates'],original_portfolio=case['joint_histories'][0]['portfolio'],
                        all_admissions=population(events,detail,calendar),
                        completed_only=population([events[i] for i in completed],[detail[i] for i in completed],calendar),
                        raw_signal_availability_clocks=clocks,event_path_details=detail))
report = dict(kind='STREAMING_BATCH18_PRE_SCREEN_METHOD_DESCRIPTIVE_SUPPLEMENT_V1',
              timing='Method already frozen before batch18 design and any batch18 outcomes; application after immutable screen.',
              original_screen_execution_git_commit=original['evaluator_git_commit'],runtime=runtime,
              supplement_execution_git_commit=subprocess.check_output(['git','rev-parse','HEAD'],cwd=root,text=True).strip(),
              input_results_sha256=sha(result_path),frozen_evaluator_sha256=sha(source),input_config_sha256=sha(config_path),
              method_bindings={key:sha(path) for key,path in bindings.items()},independent_method_review_sha256=sha(review_path),
              application_source_sha256=sha(Path(__file__)),source=evidence,candidates=reports,
              scope='Each candidate independently:40conservative sole joint history; duplicate no-protection gap events identical. All admissions and completed-only separately.',
              new_strategy_trials=0,changed_strategy_rules=False,changed_gates=False,viable=False,
              future_confirmation_started=False,orders=False)
assert sha(result_path) == report['input_results_sha256']
target.write_text(json.dumps(report,indent=2,allow_nan=False)+'\n',encoding='utf-8')
print(json.dumps(dict(output=str(target),sha256=sha(target),candidates=[dict(candidate=r['candidate'],paths=len(r['event_path_details'])) for r in reports])))
