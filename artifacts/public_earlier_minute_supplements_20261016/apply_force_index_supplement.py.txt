def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


target = base / 'outputs/ForceIndex-batch12-descriptive-supplement.json'
assert not target.exists(), 'retain previous application'
review_path = base / 'outputs/Public-screen-supplement-draft-independent-review.json'
review = json.loads(review_path.read_text())
assert not review['blocking_findings']
assert review['status'] == 'PASS_DESCRIPTIVE_DRAFT_METHOD_BEFORE_RAW_APPLICATION'
bindings = {
    'draft_evaluator_exact_sha256': base / 'work/draft_public_screen_supplement.py',
    'draft_test_exact_sha256': base / 'work/draft_test_public_screen_supplement.py',
    'method_exact_sha256': base / 'outputs/Public-screen-descriptive-supplement-method-draft.md',
}
for key, path in bindings.items():
    assert sha(path) == review[key], key

assert subprocess.check_output(['git','show','HEAD:'+index_path.relative_to(root).as_posix()],cwd=root)==index_path.read_bytes()
index=json.loads(index_path.read_text())
assert sha(result_path)==index['raw_sha256']
original=json.loads(result_path.read_text())
assert sha(source_path)==original['evaluator_source_sha256']
assert subprocess.check_output(['git','show','HEAD:'+source_path.relative_to(root).as_posix()],cwd=root).replace(b'\r\n',b'\n')==source_path.read_bytes().replace(b'\r\n',b'\n')

config_path = root / 'config/public_force_index_batch12_v1.json'
config = frozen.checked_spec(root, config_path)
runtime=dict(python=platform.python_version(),numpy=np.__version__,pandas=pd.__version__,platform=sys.platform)

    import scipy
    runtime['scipy']=scipy.__version__

panel, evidence = frozen.shared.retained_panel(
    root, base / 'outputs/Minute-continuous-source-data',

loader = importlib.util.spec_from_file_location(
    'orderflow_edge_lab.public_screen_supplement', bindings['draft_evaluator_exact_sha256'])

loader.loader.exec_module(method)
case = candidate['cost_and_gap_cases']['40:conservative_limit']

assert all(len(states) == 1 for states in case['symbols'].values())
events = [event for states in case['symbols'].values() for event in states[0]['events']]
assert len(events)==case['joint_histories'][0]['summary']['admitted']

detail = [method.excursion_details(event, panel[event['symbol']]) for event in events]
calendar = case['joint_histories'][0]['diagnostics']['calendar_weeks_with_zeros']
assert calendar == 172


def population(items, paths):
    result = method.event_summary(items, calendar)
    fields = ('pre_first_mfe_mae_bps_bounds', 'mae_through_endpoint_bps_bounds',
              'hypothetical_peak_net_after_cost_bps_bounds', 'mfe_giveback_to_endpoint_bps_bounds')
    result['path_metrics'] = {
        key: {stat: [function([p[key][i] for p in paths]) for i in (0, 1)]
              for stat, function in [('mean', method.average), ('median', method.middle)]}
        for key in fields}
    result['path_fractions'] = {
        key: sum(p[key] for p in paths) / len(paths) if paths else None
        for key in ('endpoint_correct_direction', 'definite_positive_excursion', 'possible_positive_excursion')}
    result['hypothetical_peak_clears_baseline_cost_fraction_bounds'] = [
        sum(p['hypothetical_peak_net_after_cost_bps_bounds'][i] > 0 for p in paths) / len(paths)
        if paths else None for i in (0, 1)]
    return result


clocks = {}

    feature = frozen.source_features(frame, frozen.CANDIDATES[0], config)
    clock = method.source_signal_clock(frame, feature['long'], config['warmup_bars'])
    expected = case['symbols'][symbol][0]['counts']['raw_entry_flags']
    assert clock['completion_clock']['n'] == expected, symbol
    clocks[symbol] = dict(raw_entry_flags=expected, **clock)

    input_config_sha256=sha(config_path), method_bindings={k: sha(p) for k, p in bindings.items()},
    independent_method_review_sha256=sha(review_path), application_source_sha256=sha(Path(__file__)),
    source=evidence, original_decision=candidate['decision'], original_failed_gates=candidate['failed_gates'],

    scope='40:conservative_limit sole joint history; no-protection alternative gap label has identical events. All baseline admissions and completed-only populations reported separately.',
